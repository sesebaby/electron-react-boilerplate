# InventoryService 重构实施步骤详细指南

## 🎯 重构目标回顾

将1613行的巨型InventoryService拆分为6个职责单一的轻量服务，移除复杂缓存系统，简化架构层级，符合"单机版小型进销存，不要过度工程化"的项目定位。

### 阶段二：核心服务实现（第2-4天）

#### 2.1 实现ProductService（第2天上午）

创建 `src/services/simple/ProductService.ts`：
```typescript
import { Product, ProductStatus } from '../../types/entities';
import { SimpleServiceResult, ProductFilter } from './types';
import { ElectronDatabase } from '../database/electronDatabase';
import { v4 as uuidv4 } from 'uuid';

export class ProductService {
  private db: ElectronDatabase;

  constructor() {
    this.db = new ElectronDatabase();
  }

  async getProducts(filter?: ProductFilter): Promise<SimpleServiceResult<Product[]>> {
    try {
      const products = await this.db.getAllProducts();
      let filtered = products;

      if (filter?.categoryId) {
        filtered = filtered.filter(p => p.categoryId === filter.categoryId);
      }
      if (filter?.status) {
        filtered = filtered.filter(p => p.status === filter.status);
      }
      if (filter?.search) {
        const searchLower = filter.search.toLowerCase();
        filtered = filtered.filter(p => 
          p.name.toLowerCase().includes(searchLower) ||
          p.sku.toLowerCase().includes(searchLower)
        );
      }

      return { success: true, data: filtered };
    } catch (error) {
      return { 
        success: false, 
        error: error instanceof Error ? error.message : '获取产品失败' 
      };
    }
  }

  async getProduct(id: string): Promise<SimpleServiceResult<Product>> {
    try {
      const product = await this.db.getProduct(id);
      if (!product) {
        return { success: false, error: '产品不存在' };
      }
      return { success: true, data: product };
    } catch (error) {
      return { 
        success: false, 
        error: error instanceof Error ? error.message : '获取产品失败' 
      };
    }
  }

  async createProduct(data: Omit<Product, 'id' | 'createdAt' | 'updatedAt'>): Promise<SimpleServiceResult<Product>> {
    try {
      // 基础验证
      if (!data.name?.trim()) {
        return { success: false, error: '商品名称不能为空' };
      }
      if (!data.sku?.trim()) {
        return { success: false, error: 'SKU不能为空' };
      }

      // SKU唯一性检查
      const existingProduct = await this.db.getProductBySKU(data.sku);
      if (existingProduct) {
        return { success: false, error: `SKU "${data.sku}" 已存在` };
      }

      const product: Product = {
        ...data,
        id: uuidv4(),
        createdAt: new Date(),
        updatedAt: new Date()
      };

      const result = await this.db.createProduct(product);
      return { success: true, data: result };
    } catch (error) {
      return { 
        success: false, 
        error: error instanceof Error ? error.message : '创建产品失败' 
      };
    }
  }

  async updateProduct(id: string, data: Partial<Product>): Promise<SimpleServiceResult<Product>> {
    try {
      const existingProduct = await this.db.getProduct(id);
      if (!existingProduct) {
        return { success: false, error: '产品不存在' };
      }

      // SKU唯一性检查（如果更新了SKU）
      if (data.sku && data.sku !== existingProduct.sku) {
        const duplicateProduct = await this.db.getProductBySKU(data.sku);
        if (duplicateProduct) {
          return { success: false, error: `SKU "${data.sku}" 已存在` };
        }
      }

      const updatedProduct = {
        ...existingProduct,
        ...data,
        updatedAt: new Date()
      };

      const result = await this.db.updateProduct(id, updatedProduct);
      return { success: true, data: result };
    } catch (error) {
      return { 
        success: false, 
        error: error instanceof Error ? error.message : '更新产品失败' 
      };
    }
  }

  async deleteProduct(id: string): Promise<SimpleServiceResult<boolean>> {
    try {
      const product = await this.db.getProduct(id);
      if (!product) {
        return { success: false, error: '产品不存在' };
      }

      // 检查是否有库存
      const hasStock = await this.db.checkProductHasStock(id);
      if (hasStock) {
        return { success: false, error: '商品有库存，无法删除' };
      }

      await this.db.deleteProduct(id);
      return { success: true, data: true };
    } catch (error) {
      return { 
        success: false, 
        error: error instanceof Error ? error.message : '删除产品失败' 
      };
    }
  }

  async searchProducts(searchTerm: string): Promise<SimpleServiceResult<Product[]>> {
    return this.getProducts({ search: searchTerm });
  }

  async validateSKU(sku: string, excludeId?: string): Promise<SimpleServiceResult<boolean>> {
    try {
      const existingProduct = await this.db.getProductBySKU(sku);
      const isValid = !existingProduct || (excludeId && existingProduct.id === excludeId);
      return { success: true, data: isValid };
    } catch (error) {
      return { 
        success: false, 
        error: error instanceof Error ? error.message : 'SKU验证失败' 
      };
    }
  }
}
```

#### 2.2 实现CategoryService（第2天下午）

import { Category } from '../../types/entities';
import { SimpleServiceResult } from './types';
import { ElectronDatabase } from '../database/electronDatabase';
import { v4 as uuidv4 } from 'uuid';

export class CategoryService {
  private db: ElectronDatabase;

  constructor() {
    this.db = new ElectronDatabase();
  }

  async getCategories(): Promise<SimpleServiceResult<Category[]>> {
    try {
      const categories = await this.db.getAllCategories();
      return { success: true, data: categories };
    } catch (error) {
      return { 
        success: false, 
        error: error instanceof Error ? error.message : '获取分类失败' 
      };
    }
  }

  async getCategory(id: string): Promise<SimpleServiceResult<Category>> {
    try {
      const category = await this.db.getCategory(id);
      if (!category) {
        return { success: false, error: '分类不存在' };
      }
      return { success: true, data: category };
    } catch (error) {
      return { 
        success: false, 
        error: error instanceof Error ? error.message : '获取分类失败' 
      };
    }
  }

  async createCategory(data: Omit<Category, 'id' | 'createdAt' | 'updatedAt'>): Promise<SimpleServiceResult<Category>> {
    try {
      if (!data.name?.trim()) {
        return { success: false, error: '分类名称不能为空' };
      }

      const category: Category = {
        ...data,
        id: uuidv4(),
        createdAt: new Date(),
        updatedAt: new Date()
      };

      const result = await this.db.createCategory(category);
      return { success: true, data: result };
    } catch (error) {
      return { 
        success: false, 
        error: error instanceof Error ? error.message : '创建分类失败' 
      };
    }
  }

  async updateCategory(id: string, data: Partial<Category>): Promise<SimpleServiceResult<Category>> {
    try {
      const existingCategory = await this.db.getCategory(id);
      if (!existingCategory) {
        return { success: false, error: '分类不存在' };
      }

      const updatedCategory = {
        ...existingCategory,
        ...data,
        updatedAt: new Date()
      };

      const result = await this.db.updateCategory(id, updatedCategory);
      return { success: true, data: result };
    } catch (error) {
      return { 
        success: false, 
        error: error instanceof Error ? error.message : '更新分类失败' 
      };
    }
  }

  async deleteCategory(id: string): Promise<SimpleServiceResult<boolean>> {
    try {
      const category = await this.db.getCategory(id);
      if (!category) {
        return { success: false, error: '分类不存在' };
      }

      // 检查是否有产品使用此分类
      const productCount = await this.db.getProductCountByCategory(id);
      if (productCount > 0) {
        return { success: false, error: `该分类下有 ${productCount} 个产品，无法删除` };
      }

      await this.db.deleteCategory(id);
      return { success: true, data: true };
    } catch (error) {
      return { 
        success: false, 
        error: error instanceof Error ? error.message : '删除分类失败' 
      };
    }
  }

  async checkCategoryUsage(id: string): Promise<SimpleServiceResult<{ productCount: number }>> {
    try {
      const productCount = await this.db.getProductCountByCategory(id);
      return { success: true, data: { productCount } };
    } catch (error) {
      return { 
        success: false, 
        error: error instanceof Error ? error.message : '检查分类使用情况失败' 
      };
    }
  }
}
